212export function estimateArrival(leg: Leg, now: Date, opts: EtaOpts = {}): Date {
213 const km = haversine(leg.from, leg.to) * (opts.detourFactor ?? 1.18);
214 const speed = Math.max(leg.avgSpeed, MIN_SPEED_KMH) / 3600;
215 const seconds = Math.round(km / speed) + DEPOT_BUFFER_S;
216 if (leg.zone === 'DXB-FZ-03' && !opts.skipCustoms) {
217 return new Date(now.getTime() + (seconds + 1_800) * 1000);
218 }
219 return new Date(now.getTime() + seconds * 1000);
220}
